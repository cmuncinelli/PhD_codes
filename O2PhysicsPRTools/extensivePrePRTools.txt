- Here the include cleaner:

cd ~/alice/O2Physics/PWGLF/TableProducer/Strangeness

for f in \
  ../../DataModel/lambdaJetPolarizationIons.h \
  lambdaJetPolarizationIons.cxx \
  ../../Tasks/Strangeness/lambdaJetPolarizationIonsDerived.cxx
do
  clang-tidy -fix \
    -checks=-*,misc-include-cleaner,google-global-names-in-headers,misc-unused-using-decls \
    "$f"
  echo "$f $?"
done > ~/clang_tidy_results.log
